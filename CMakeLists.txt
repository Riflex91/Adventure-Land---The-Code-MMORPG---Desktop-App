cmake_minimum_required(VERSION 3.21)
project(AdventureLandReferenceOSDesktop VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

find_package(SQLite3 REQUIRED)
set(SOURCES src/main.cpp src/app.cpp src/database.cpp)
if(WIN32)
  list(APPEND SOURCES src/platform_windows.cpp)
else()
  find_package(X11 REQUIRED)
  list(APPEND SOURCES src/platform_linux_x11.cpp)
endif()
add_executable(adventure-land-reference-os ${SOURCES})
target_include_directories(adventure-land-reference-os PRIVATE src)
target_link_libraries(adventure-land-reference-os PRIVATE SQLite::SQLite3)
if(WIN32)
  target_compile_definitions(adventure-land-reference-os PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(adventure-land-reference-os PRIVATE user32 gdi32)
else()
  target_include_directories(adventure-land-reference-os PRIVATE ${X11_INCLUDE_DIR})
  target_link_libraries(adventure-land-reference-os PRIVATE ${X11_LIBRARIES})
endif()
if(MSVC)
  target_compile_options(adventure-land-reference-os PRIVATE /W4 /permissive- /EHsc)
else()
  target_compile_options(adventure-land-reference-os PRIVATE -Wall -Wextra -Wpedantic)
endif()
install(TARGETS adventure-land-reference-os RUNTIME DESTINATION .)
install(DIRECTORY resources DESTINATION .)

option(AL_BUILD_TESTS "Build native core tests" ON)
if(AL_BUILD_TESTS)
  add_executable(al-core-test tests/core_test.cpp src/database.cpp)
  target_include_directories(al-core-test PRIVATE src)
  target_link_libraries(al-core-test PRIVATE SQLite::SQLite3)
  if(MSVC)
    target_compile_options(al-core-test PRIVATE /W4 /permissive- /EHsc)
  else()
    target_compile_options(al-core-test PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endif()
